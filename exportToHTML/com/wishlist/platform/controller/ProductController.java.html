1    package com.wishlist.platform.controller;
2    
3    import com.wishlist.platform.entity.Product;
4    import com.wishlist.platform.repository.ProductRepository;
5    import org.springframework.http.ResponseEntity;
6    import org.springframework.web.bind.annotation.*;
7    
8    import java.util.List;
9    
10   @RestController
11   @RequestMapping("/api/products")
12   public class ProductController {
13   
14       private final ProductRepository productRepository;
15   
16       public ProductController(ProductRepository productRepository) {
17           this.productRepository = productRepository;
18       }
19   
20       @GetMapping
21       public List<Product> getAllProducts() {
22           return productRepository.findAll();
23       }
24   
25       @GetMapping("/{id}")
26       public ResponseEntity<Product> getProductById(@PathVariable Long id) {
27           return productRepository.findById(id)
28                   .map(ResponseEntity::ok)
29                   .orElse(ResponseEntity.notFound().build());
30       }
31   
32       @PostMapping
33       public ResponseEntity<Product> addProduct(@RequestBody Product product) {
34           Product savedProduct = productRepository.save(product);
35           return ResponseEntity.ok(savedProduct);
36       }
37   
38       @DeleteMapping("/{id}")
39       public ResponseEntity<Void> deleteProduct(@PathVariable Long id) {
40           productRepository.deleteById(id);
41           return ResponseEntity.noContent().build();
42       }
43   }
